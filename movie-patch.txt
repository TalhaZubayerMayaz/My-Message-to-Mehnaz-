/* ---- movie tickets: swipe to tear ---- */
let mtDone = 0;
function enterMTicket(){
  mtDone = 0;
  document.querySelectorAll('#scene-mticket .mt').forEach(t=> t.classList.remove('torn'));
  document.getElementById('mtGo').style.display = 'none';
  document.getElementById('mtHint').textContent = 'Drag along the dotted line to tear each ticket';
  showScene('scene-mticket');
}
document.querySelectorAll('#scene-mticket .mt').forEach(t=>{
  const st = t.querySelector('.stub'); let x0 = null;
  st.addEventListener('pointerdown', e=>{ if(t.classList.contains('torn')) return; x0 = e.clientX; st.setPointerCapture(e.pointerId); st.style.transition = 'none'; });
  st.addEventListener('pointermove', e=>{
    if(x0 === null) return;
    const p = Math.max(0, Math.min(1, (e.clientX - x0) / (st.offsetWidth * .8)));
    st.style.transform = `rotate(${p*7}deg)`;
    if(p >= 1){ x0 = null; st.style.transition = ''; st.style.transform = ''; t.classList.add('torn');
      const r = t.getBoundingClientRect(); spawnJoyBurst(r.left + r.width/2, r.bottom - 20, 8);
      if(++mtDone >= 2){ document.getElementById('mtGo').style.display = ''; document.getElementById('mtHint').textContent = 'Both torn ♥ our seats are waiting'; } }
  });
  const rel = ()=>{ if(x0 === null) return; x0 = null; st.style.transition = ''; st.style.transform = ''; };
  st.addEventListener('pointerup', rel); st.addEventListener('pointercancel', rel);
});
document.getElementById('mtGo').addEventListener('click', ()=> nextStep());

function enterCinema(){
  drawGrid(document.getElementById('cinE'), BACK_E, {H:'#3b2418', S:'#fbd9c3', J:'#a8702f'});
